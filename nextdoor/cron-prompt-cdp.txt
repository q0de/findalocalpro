import json,sys,asyncio,websockets
async def main():

    target = next((t[\"id\"] for t in tabs if \"nextdoor\" in t.get(\"url\",\"\")), None)
    if not target: print(\"No nextdoor tab\"); return

        r = json.loads(await ws.recv())
        print(r[\"result\"][\"result\"][\"value\"][:20000])
asyncio.run(main())
'", timeout=30
